```python
def _run_gated_steps(
    row: HindsightRecipeRow,
    params: dict[str, str],
    step_defs: tuple[dict[str, Any], ...],
    *,
    fail_prefix: str,
) -> ProviderRecipeResult | None:
    """Run declared steps behind the row's source-verification gate.

    Returns a failure ProviderRecipeResult if the gate blocks execution or the
    step sequence fails; returns None when the caller should fall through to
    its own success status (steps ran clean, or there were no steps to run).
    """
    if not step_defs:
        return None
    if row.source_status != "verified":
        return ProviderRecipeResult.fail(
            f"plugin installer source {row.source_status}; refusing to execute",
            action_needed=row.notes or row.audia_action,
        )
    steps = steps_from_defs(list(step_defs), params)
    seq = run_steps(steps, {}, fail_prefix=fail_prefix)
    return seq if not seq.success else None
```
