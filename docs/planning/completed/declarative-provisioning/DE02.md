# Add a generic delete-path/remove-tree step so recipe uninstall can remove owned files and dirs

## Description

The declarative recipe engine can create files (write-file, download, config-set) but cannot delete a file or directory on uninstall. Consequence: hindsight-codex.yaml uninstall only config-removes codex.json keys and leaves the fetched scripts under ~/.hindsight/codex/scripts/ behind (INSTALL_MAP contract says uninstall must remove downloaded scripts). Add a generic, self-reverting delete-path step to foundation/steps, in the same spirit as config-remove being the counterpart of config-set.

## Detailed Solution & Technical Design

New foundation step type 'delete-path' (foundation/steps/structured.py + factory registration + co-located schema fragment). Fields: path (required), recursive (bool, default false). run(): if path exists, capture enough to restore on compensate (for a single file: prior content; for a tree: this is best-effort - document that recursive delete compensation is not guaranteed and is intended for capability-owned caches like ~/.hindsight/codex/scripts). plan(): describe. This stays domain-neutral (no provider/hindsight knowledge). Then hindsight-codex.yaml uninstall-steps add a delete-path for ~/.hindsight/codex/scripts (and optionally the codex dir).

=== foundation/steps/structured.py ===
import shutil

class DeletePathStep:
    """Delete a capability-owned file or directory; best-effort self-revert.

    A single file is restored on compensate from captured BYTES (binary-safe). A
    recursive directory delete is NOT restorable (documented) - use only for
    caches the capability wholly owns (e.g. ~/.hindsight/codex/scripts).
    """
    def __init__(self, id, path, *, recursive=False, registry=None, recipe_id=None):
        self.id = id; self.path = path; self.recursive = recursive
        self.registry = registry; self.recipe_id = recipe_id
        self._prior_bytes = None; self._removed = False

    def run(self, context):
        target = Path(self.path).expanduser()
        if not target.exists():
            return StepResult(status="ok", outputs={"path": self.path, "existed": False})
        if target.is_dir():
            if not self.recursive:
                return StepResult(status="failed", reason=f"{target} is a directory; set recursive: true")
            shutil.rmtree(target); self._prior_bytes = None
        else:
            self._prior_bytes = target.read_bytes()   # RV828 D1: binary-safe
            target.unlink()
        self._removed = True
        return StepResult(status="ok", outputs={"path": self.path, "existed": True})

    def compensate(self, context):
        if not self._removed:
            return StepResult(status="skipped", reason="run never removed anything")
        if self._prior_bytes is not None:            # single file restore
            t = Path(self.path).expanduser(); t.parent.mkdir(parents=True, exist_ok=True)
            t.write_bytes(self._prior_bytes)
            return StepResult(status="ok", outputs={"path": self.path, "restored": True})
        return StepResult(status="skipped", reason="recursive delete is not restorable")

    def plan(self, context):
        return StepResult(status="planned", outputs={"path": self.path, "recursive": self.recursive})

def _build_delete_path(data, params=None):
    path = data["path"]
    if params is not None:
        path = strict_substitute(path, params, f"step.{data.get('id','delete-path')}.path")
    return DeletePathStep(id=data.get("id","delete-path"), path=path,
